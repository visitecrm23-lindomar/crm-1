// GET - Listar viagens
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const search = searchParams.get('search') || ''
    const status = searchParams.get('status') || ''
    const type = searchParams.get('type') || ''

    const skip = (page - 1) * limit

    const where: any = {
      tenant_id: session.user.tenant_id,
    }

    if (search) {
      where.OR = [
        { title: { contains: search, mode: 'insensitive' } },
        { destination: { contains: search, mode: 'insensitive' } },
      ]
    }

    if (status) {
      where.status = status
    }

    if (type) {
      where.type = type
    }

    const [trips, total] = await Promise.all([
      prisma.trip.findMany({
        where,
        include: {
          vehicle: true,
          accommodation: true,
          _count: {
            select: {
              reservations: true,
            },
          },
        },
        orderBy: {
          departure_date: 'asc',
        },
        skip,
        take: limit,
      }),
      prisma.trip.count({ where }),
    ])

    return NextResponse.json({
      trips,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error) {
    console.error('Erro ao buscar viagens:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar viagens' },
      { status: 500 }
    )
  }
}

// POST - Criar viagem
export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const body = await req.json()
    const data = tripSchema.parse(body)

    // Gerar slug único
    const baseSlug = data.title
      .toLowerCase()
      .normalize('NFD')
      .replace(/[\u0300-\u036f]/g, '')
      .replace(/[^a-z0-9]+/g, '-')
      .replace(/(^-|-$)/g, '')

    let slug = baseSlug
    let counter = 1
    while (
      await prisma.trip.findFirst({
        where: {
          tenant_id: session.user.tenant_id,
          slug,
        },
      })
    ) {
      slug = `${baseSlug}-${counter}`
      counter++
    }

    // Inicializar mapa de assentos
    const seatMap = generateSeatMap(data.total_seats, data.bus_layout)

    const trip = await prisma.trip.create({
      data: {
        ...data,
        slug,
        tenant_id: session.user.tenant_id,
        departure_date: new Date(data.departure_date),
        return_date: data.return_date ? new Date(data.return_date) : null,
        seat_map: seatMap,
      },
    })

    return NextResponse.json(trip, { status: 201 })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Dados inválidos', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Erro ao criar viagem:', error)
    return NextResponse.json(
      { error: 'Erro ao criar viagem' },
      { status: 500 }
    )
  }
}

// Função auxiliar para gerar mapa de assentos
function generateSeatMap(totalSeats: number, layout: string) {
  const seats = []
  const [left, right] = layout.split('x').map(Number)
  const seatsPerRow = left + right

  for (let i = 1; i <= totalSeats; i++) {
    seats.push({
      number: i,
      status: 'available', // available, reserved, confirmed, blocked
      passenger: null,
      reservation_id: null,
    })
  }

  return {
    layout,
    total: totalSeats,
    seatsPerRow,
    seats,
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

// GET - Buscar mapa de assentos
export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const trip = await prisma.trip.findFirst({
      where: {
        id: params.id,
        tenant_id: session.user.tenant_id,
      },
      select: {
        id: true,
        title: true,
        departure_date: true,
        total_seats: true,
        bus_layout: true,
        seat_map: true,
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    return NextResponse.json(trip)
  } catch (error) {
    console.error('Erro ao buscar mapa de assentos:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar mapa de assentos' },
      { status: 500 }
    )
  }
}

// PATCH - Atualizar assento
export async function PATCH(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const body = await req.json()
    const { seatNumber, clientId, clientName, clientPhone, boardingPoint } = body

    const trip = await prisma.trip.findFirst({
      where: {
        id: params.id,
        tenant_id: session.user.tenant_id,
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    const seatMap: any = trip.seat_map
    const seatIndex = seatMap.seats.findIndex(
      (s: any) => s.number === seatNumber
    )

    if (seatIndex === -1) {
      return NextResponse.json(
        { error: 'Assento não encontrado' },
        { status: 404 }
      )
    }

    // Atualizar assento
    seatMap.seats[seatIndex] = {
      ...seatMap.seats[seatIndex],
      status: 'reserved',
      passenger: {
        client_id: clientId,
        name: clientName,
        phone: clientPhone,
        boarding_point: boardingPoint,
      },
    }

    // Salvar no banco
    const updatedTrip = await prisma.trip.update({
      where: { id: params.id },
      data: { seat_map: seatMap },
    })

    // TODO: Emitir evento WebSocket para atualização em tempo real
    // await emitSeatUpdate(params.id, seatMap)

    return NextResponse.json(updatedTrip.seat_map)
  } catch (error) {
    console.error('Erro ao atualizar assento:', error)
    return NextResponse.json(
      { error: 'Erro ao atualizar assento' },
      { status: 500 }
    )
  }
}

'use client'

import { useState, useEffect } from 'react'
import { Button } from '@/components/ui/button'
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog'
import { Input } from '@/components/ui/input'
import { Label } from '@/components/ui/label'
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select'
import { useToast } from '@/components/ui/use-toast'

interface Seat {
  number: number
  status: 'available' | 'reserved' | 'confirmed' | 'blocked'
  passenger: {
    client_id?: string
    name: string
    phone: string
    boarding_point: string
  } | null
}

interface SeatMapProps {
  tripId: string
}

export function SeatMap({ tripId }: SeatMapProps) {
  const { toast } = useToast()
  const [seatMap, setSeatMap] = useState<any>(null)
  const [selectedSeat, setSelectedSeat] = useState<number | null>(null)
  const [isModalOpen, setIsModalOpen] = useState(false)
  const [isLoading, setIsLoading] = useState(true)

  useEffect(() => {
    loadSeatMap()
  }, [tripId])

  async function loadSeatMap() {
    try {
      const res = await fetch(`/api/trips/${tripId}/seat-map`)
      const data = await res.json()
      setSeatMap(data.seat_map)
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao carregar mapa de assentos',
        variant: 'destructive',
      })
    } finally {
      setIsLoading(false)
    }
  }

  function handleSeatClick(seatNumber: number) {
    const seat = seatMap.seats.find((s: Seat) => s.number === seatNumber)
    
    if (seat.status === 'available' || seat.status === 'reserved') {
      setSelectedSeat(seatNumber)
      setIsModalOpen(true)
    }
  }

  async function handleSaveSeat(formData: any) {
    try {
      const res = await fetch(`/api/trips/${tripId}/seat-map`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          seatNumber: selectedSeat,
          ...formData,
        }),
      })

      if (!res.ok) throw new Error()

      const updatedSeatMap = await res.json()
      setSeatMap(updatedSeatMap)
      setIsModalOpen(false)
      
      toast({
        title: 'Sucesso',
        description: 'Assento atualizado com sucesso',
      })
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao atualizar assento',
        variant: 'destructive',
      })
    }
  }

  if (isLoading) {
    return <div>Carregando mapa de assentos...</div>
  }

  if (!seatMap) {
    return <div>Mapa de assentos não disponível</div>
  }

  const [left, right] = seatMap.layout.split('x').map(Number)
  const rows = Math.ceil(seatMap.total / seatMap.seatsPerRow)

  return (
    <div className="space-y-6">
      {/* Legenda */}
      <div className="flex gap-4 justify-center">
        <div className="flex items-center gap-2">
          <div className="w-8 h-8 bg-white border-2 border-gray-300 rounded" />
          <span className="text-sm">Disponível</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-8 h-8 bg-orange-200 border-2 border-orange-400 rounded" />
          <span className="text-sm">Reservado</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-8 h-8 bg-green-200 border-2 border-green-400 rounded" />
          <span className="text-sm">Confirmado</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="w-8 h-8 bg-gray-300 border-2 border-gray-400 rounded" />
          <span className="text-sm">Bloqueado</span>
        </div>
      </div>

      {/* Resumo */}
      <div className="bg-gray-100 p-4 rounded-lg">
        <div className="grid grid-cols-3 gap-4 text-center">
          <div>
            <div className="text-2xl font-bold">{seatMap.total}</div>
            <div className="text-sm text-gray-600">Total</div>
          </div>
          <div>
            <div className="text-2xl font-bold text-green-600">
              {seatMap.seats.filter((s: Seat) => 
                s.status === 'reserved' || s.status === 'confirmed'
              ).length}
            </div>
            <div className="text-sm text-gray-600">Ocupados</div>
          </div>
          <div>
            <div className="text-2xl font-bold text-blue-600">
              {seatMap.seats.filter((s: Seat) => 
                s.status === 'available'
              ).length}
            </div>
            <div className="text-sm text-gray-600">Disponíveis</div>
          </div>
        </div>
      </div>

      {/* Mapa do Ônibus */}
      <div className="bg-gray-50 p-8 rounded-lg">
        <div className="max-w-md mx-auto">
          {/* Motorista */}
          <div className="mb-4 flex justify-end">
            <div className="w-12 h-12 bg-gray-700 rounded-full flex items-center justify-center text-white">
              🚗
            </div>
          </div>

          {/* Assentos */}
          <div className="space-y-2">
            {Array.from({ length: rows }).map((_, rowIndex) => {
              const startSeat = rowIndex * seatMap.seatsPerRow + 1
              const rowSeats = seatMap.seats.slice(
                startSeat - 1,
                startSeat - 1 + seatMap.seatsPerRow
              )

              return (
                <div key={rowIndex} className="flex gap-2 justify-center">
                  {/* Lado esquerdo */}
                  {rowSeats.slice(0, left).map((seat: Seat) => (
                    <SeatButton
                      key={seat.number}
                      seat={seat}
                      onClick={() => handleSeatClick(seat.number)}
                    />
                  ))}

                  {/* Corredor */}
                  <div className="w-8" />

                  {/* Lado direito */}
                  {rowSeats.slice(left).map((seat: Seat) => (
                    <SeatButton
                      key={seat.number}
                      seat={seat}
                      onClick={() => handleSeatClick(seat.number)}
                    />
                  ))}
                </div>
              )
            })}
          </div>
        </div>
      </div>

      {/* Modal de Seleção */}
      <SeatSelectionModal
        isOpen={isModalOpen}
        onClose={() => setIsModalOpen(false)}
        seatNumber={selectedSeat}
        onSave={handleSaveSeat}
      />
    </div>
  )
}

function SeatButton({ seat, onClick }: { seat: Seat; onClick: () => void }) {
  const colors = {
    available: 'bg-white border-gray-300 hover:bg-blue-50',
    reserved: 'bg-orange-200 border-orange-400',
    confirmed: 'bg-green-200 border-green-400',
    blocked: 'bg-gray-300 border-gray-400 cursor-not-allowed',
  }

  return (
    <button
      onClick={onClick}
      disabled={seat.status === 'blocked'}
      className={`
        w-12 h-12 border-2 rounded flex flex-col items-center justify-center
        text-xs transition-colors
        ${colors[seat.status]}
      `}
    >
      <div className="font-bold">{seat.number}</div>
      {seat.passenger && (
        <div className="text-[8px] truncate w-full px-1">
          {seat.passenger.name.split(' ')[0]}
        </div>
      )}
    </button>
  )
}

function generateCSV(trip: any, passengers: any[]) {

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { z } from 'zod'

const passengerSchema = z.object({
  nome: z.string(),
  cpf: z.string().optional(),
  data_nascimento: z.string().optional(),
  tipo: z.string(), // adulto, crianca, idoso, bebe
  assento: z.number(),
  embarque: z.string(),
  pacote: z.string().optional(),
})

const reservationSchema = z.object({
  client_id: z.string(),
  trip_id: z.string(),
  passengers: z.array(passengerSchema),
  seats: z.array(z.number()),
  total_value: z.number(),
  paid_value: z.number().default(0),
  discount: z.number().optional(),
  payment_method: z.string().optional(),
  installments: z.number().optional(),
  seller_id: z.string().optional(),
  commission_percentage: z.number().optional(),
  observations: z.string().optional(),
})

// GET - Listar reservas
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const page = parseInt(searchParams.get('page') || '1')
    const limit = parseInt(searchParams.get('limit') || '20')
    const status = searchParams.get('status') || ''
    const tripId = searchParams.get('trip_id') || ''

    const skip = (page - 1) * limit

    const where: any = {
      tenant_id: session.user.tenant_id,
    }

    // Se vendedor, mostrar apenas suas reservas
    if (session.user.role === 'vendedor') {
      where.seller_id = session.user.id
    }

    if (status) {
      where.status = status
    }

    if (tripId) {
      where.trip_id = tripId
    }

    const [reservations, total] = await Promise.all([
      prisma.reservation.findMany({
        where,
        include: {
          client: true,
          trip: {
            select: {
              id: true,
              title: true,
              departure_date: true,
            },
          },
          seller: {
            select: {
              id: true,
              name: true,
            },
          },
        },
        orderBy: {
          created_at: 'desc',
        },
        skip,
        take: limit,
      }),
      prisma.reservation.count({ where }),
    ])

    return NextResponse.json({
      reservations,
      pagination: {
        page,
        limit,
        total,
        totalPages: Math.ceil(total / limit),
      },
    })
  } catch (error) {
    console.error('Erro ao buscar reservas:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar reservas' },
      { status: 500 }
    )
  }
}

// POST - Criar reserva
export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const body = await req.json()
    const data = reservationSchema.parse(body)

    // Verificar se viagem existe e pertence ao tenant
    const trip = await prisma.trip.findFirst({
      where: {
        id: data.trip_id,
        tenant_id: session.user.tenant_id,
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    // Verificar se assentos estão disponíveis
    const seatMap: any = trip.seat_map
    for (const seatNumber of data.seats) {
      const seat = seatMap.seats.find((s: any) => s.number === seatNumber)
      if (!seat || seat.status !== 'available') {
        return NextResponse.json(
          { error: `Assento ${seatNumber} não está disponível` },
          { status: 400 }
        )
      }
    }

    // Calcular saldo
    const balance = data.total_value - data.paid_value

    // Calcular comissão
    let commissionValue = 0
    if (data.commission_percentage) {
      commissionValue = (data.total_value * data.commission_percentage) / 100
    }

    // Gerar código do voucher
    const voucherCode = generateVoucherCode()

    // Criar reserva em uma transação
    const reservation = await prisma.$transaction(async (tx) => {
      // Criar reserva
      const newReservation = await tx.reservation.create({
        data: {
          tenant_id: session.user.tenant_id,
          client_id: data.client_id,
          trip_id: data.trip_id,
          passengers: data.passengers,
          seats: data.seats,
          total_value: data.total_value,
          paid_value: data.paid_value,
          balance,
          discount: data.discount,
          payment_method: data.payment_method,
          installments: data.installments,
          installment_value: data.installments
            ? data.total_value / data.installments
            : null,
          seller_id: data.seller_id || session.user.id,
          commission_percentage: data.commission_percentage,
          commission_value: commissionValue,
          voucher_code: voucherCode,
          status: data.paid_value >= data.total_value ? 'confirmada' : 'pendente',
          observations: data.observations,
        },
      })

      // Atualizar mapa de assentos
      for (let i = 0; i < data.seats.length; i++) {
        const seatNumber = data.seats[i]
        const passenger = data.passengers[i]
        const seatIndex = seatMap.seats.findIndex(
          (s: any) => s.number === seatNumber
        )
        seatMap.seats[seatIndex] = {
          ...seatMap.seats[seatIndex],
          status: 'reserved',
          passenger: {
            name: passenger.nome,
            cpf: passenger.cpf,
            boarding_point: passenger.embarque,
          },
          reservation_id: newReservation.id,
        }
      }

      await tx.trip.update({
        where: { id: data.trip_id },
        data: { seat_map: seatMap },
      })

      // Registrar pagamento se houver valor pago
      if (data.paid_value > 0) {
        await tx.payment.create({
          data: {
            reservation_id: newReservation.id,
            amount: data.paid_value,
            payment_method: data.payment_method || 'dinheiro',
            payment_date: new Date(),
          },
        })
      }

      // Criar comissão se houver
      if (commissionValue > 0 && data.seller_id) {
        await tx.commission.create({
          data: {
            tenant_id: session.user.tenant_id,
            seller_id: data.seller_id,
            reservation_id: newReservation.id,
            sale_value: data.total_value,
            commission_percentage: data.commission_percentage!,
            commission_value: commissionValue,
            status: 'pendente',
          },
        })
      }

      // Atualizar total gasto do cliente
      await tx.client.update({
        where: { id: data.client_id },
        data: {
          total_spent: {
            increment: data.total_value,
          },
        },
      })

      return newReservation
    })

    return NextResponse.json(reservation, { status: 201 })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Dados inválidos', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Erro ao criar reserva:', error)
    return NextResponse.json(
      { error: 'Erro ao criar reserva' },
      { status: 500 }
    )
  }
}

function generateVoucherCode() {
  const chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'
  let code = ''
  for (let i = 0; i < 8; i++) {
    code += chars.charAt(Math.floor(Math.random() * chars.length))
  }
  return code
}

// GET - Listar transações
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const type = searchParams.get('type') || ''
    const status = searchParams.get('status') || ''
    const startDate = searchParams.get('start_date') || ''
    const endDate = searchParams.get('end_date') || ''

    const where: any = {
      tenant_id: session.user.tenant_id,
    }

    if (type) where.type = type
    if (status) where.status = status
    if (startDate && endDate) {
      where.due_date = {
        gte: new Date(startDate),
        lte: new Date(endDate),
      }
    }

    const transactions = await prisma.transaction.findMany({
      where,
      orderBy: { due_date: 'desc' },
    })

    // Calcular totais
    const summary = {
      total_income: transactions
        .filter((t) => t.type === 'income')
        .reduce((sum, t) => sum + Number(t.amount), 0),
      total_expense: transactions
        .filter((t) => t.type === 'expense')
        .reduce((sum, t) => sum + Number(t.amount), 0),
      pending_income: transactions
        .filter((t) => t.type === 'income' && t.status === 'pendente')
        .reduce((sum, t) => sum + Number(t.amount), 0),
      pending_expense: transactions
        .filter((t) => t.type === 'expense' && t.status === 'pendente')
        .reduce((sum, t) => sum + Number(t.amount), 0),
    }

    summary.balance = summary.total_income - summary.total_expense

    return NextResponse.json({ transactions, summary })
  } catch (error) {
    console.error('Erro ao buscar transações:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar transações' },
      { status: 500 }
    )
  }
}

// POST - Criar transação
export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const body = await req.json()
    const data = transactionSchema.parse(body)

    const transaction = await prisma.transaction.create({
      data: {
        ...data,
        tenant_id: session.user.tenant_id,
        due_date: new Date(data.due_date),
        payment_date: data.payment_date ? new Date(data.payment_date) : null,
      },
    })

    return NextResponse.json(transaction, { status: 201 })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Dados inválidos', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Erro ao criar transação:', error)
    return NextResponse.json(
      { error: 'Erro ao criar transação' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { startOfMonth, endOfMonth, subMonths, format } from 'date-fns'

// GET - Relatório de fluxo de caixa
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const months = parseInt(searchParams.get('months') || '6')

    // Buscar dados dos últimos N meses
    const data = []
    for (let i = months - 1; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      // Entradas (receitas de reservas)
      const income = await prisma.reservation.aggregate({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: {
            gte: start,
            lte: end,
          },
        },
        _sum: {
          total_value: true,
        },
      })

      // Saídas (despesas)
      const expenses = await prisma.expense.aggregate({
        where: {
          tenant_id: session.user.tenant_id,
          due_date: {
            gte: start,
            lte: end,
          },
        },
        _sum: {
          amount: true,
        },
      })

      data.push({
        month: format(date, 'MMM/yyyy'),
        income: Number(income._sum.total_value || 0),
        expenses: Number(expenses._sum.amount || 0),
        balance:
          Number(income._sum.total_value || 0) -
          Number(expenses._sum.amount || 0),
      })
    }

    return NextResponse.json(data)
  } catch (error) {
    console.error('Erro ao gerar relatório:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar relatório' },
      { status: 500 }
    )
  }
}

## app/api/finance/reports/trip/[id]/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'

// GET - Relatório financeiro por viagem
export async function GET(
  req: Request,
  { params }: { params: { id: string } }
) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const trip = await prisma.trip.findFirst({
      where: {
        id: params.id,
        tenant_id: session.user.tenant_id,
      },
      include: {
        reservations: true,
        expenses: true,
      },
    })

    if (!trip) {
      return NextResponse.json(
        { error: 'Viagem não encontrada' },
        { status: 404 }
      )
    }

    // Calcular receitas
    const totalRevenue = trip.reservations.reduce(
      (sum, r) => sum + Number(r.total_value),
      0
    )
    const totalPaid = trip.reservations.reduce(
      (sum, r) => sum + Number(r.paid_value),
      0
    )
    const totalPending = totalRevenue - totalPaid

    // Calcular despesas por categoria
    const expensesByCategory = trip.expenses.reduce((acc: any, expense) => {
      if (!acc[expense.category]) {
        acc[expense.category] = {
          category: expense.category,
          total: 0,
          items: [],
        }
      }
      acc[expense.category].total += Number(expense.amount)
      acc[expense.category].items.push({
        description: expense.description,
        amount: Number(expense.amount),
      })
      return acc
    }, {})

    const totalExpenses = trip.expenses.reduce(
      (sum, e) => sum + Number(e.amount),
      0
    )

    // Calcular lucro
    const profit = totalRevenue - totalExpenses
    const margin = totalRevenue > 0 ? (profit / totalRevenue) * 100 : 0

    return NextResponse.json({
      trip: {
        id: trip.id,
        title: trip.title,
        departure_date: trip.departure_date,
      },
      revenue: {
        total: totalRevenue,
        paid: totalPaid,
        pending: totalPending,
        reservations_count: trip.reservations.length,
      },
      expenses: {
        total: totalExpenses,
        by_category: Object.values(expensesByCategory),
      },
      profit: {
        value: profit,
        margin: margin.toFixed(2),
      },
    })
  } catch (error) {
    console.error('Erro ao gerar relatório:', error)
    return NextResponse.json(
      { error: 'Erro ao gerar relatório' },
      { status: 500 }
    )
  }
}

import axios from 'axios'

const EVOLUTION_API_URL = process.env.EVOLUTION_API_URL
const EVOLUTION_API_KEY = process.env.EVOLUTION_API_KEY
const EVOLUTION_INSTANCE_NAME = process.env.EVOLUTION_INSTANCE_NAME

export class WhatsAppService {
  private api = axios.create({
    baseURL: EVOLUTION_API_URL,
    headers: {
      'Content-Type': 'application/json',
      apikey: EVOLUTION_API_KEY,
    },
  })

  async sendText(to: string, message: string) {
    try {
      const response = await this.api.post(
        `/message/sendText/${EVOLUTION_INSTANCE_NAME}`,
        {
          number: to.replace(/\D/g, ''),
          text: message,
        }
      )
      return response.data
    } catch (error) {
      console.error('Erro ao enviar mensagem WhatsApp:', error)
      throw error
    }
  }

    try {
      const response = await this.api.post(
        `/message/sendMedia/${EVOLUTION_INSTANCE_NAME}`,
        {
          number: to.replace(/\D/g, ''),
          mediaUrl,
          caption,
        }
      )
      return response.data
    } catch (error) {
      console.error('Erro ao enviar mídia WhatsApp:', error)
      throw error
    }
  }

    try {
      const response = await this.api.get(
        `/chat/findMessages/${EVOLUTION_INSTANCE_NAME}`,

      )
      return response.data
    } catch (error) {
      console.error('Erro ao buscar mensagens WhatsApp:', error)
      throw error
    }

    } catch (error) {
      console.error('Erro ao verificar status da instância:', error)

}

const transporter = nodemailer.createTransport({

export class EmailService {
  async sendEmail({
    to,
    subject,
    html,
    text,
    attachments,
  }: {
    to: string
    subject: string
    html?: string
    text?: string
    attachments?: any[]
  }) {

      const info = await transporter.sendMail({
        from: process.env.SMTP_FROM,
        to,
        subject,
        text,
        html,
        attachments,
      })

      console.log('Email enviado:', info.messageId)
      return info
    } catch (error) {
      console.error('Erro ao enviar email:', error)
      throw error
    }
  }

  async sendTemplate({
    to,
    template,
    variables,
  }: {
    to: string
    template: {
      subject: string
      content: string
    }
    variables: Record<string, string>
  }) {
    let content = template.content
    let subject = template.subject

    // Substituir variáveis
    Object.entries(variables).forEach(([key, value]) => {
      const regex = new RegExp(`{{${key}}}`, 'g')
      content = content.replace(regex, value)
      subject = subject.replace(regex, value)
    })

    return this.sendEmail({
      to,
      subject,
      html: content,
    })
  }
}

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { z } from 'zod'

const messageSchema = z.object({
  client_id: z.string().optional(),
  channel: z.enum(['whatsapp', 'email', 'instagram', 'telegram', 'interno']),
  subject: z.string().optional(),
  message: z.string(),
  attachments: z.array(z.string()).default([]),
})

// GET - Listar mensagens
export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const clientId = searchParams.get('client_id') || ''
    const channel = searchParams.get('channel') || ''

    const where: any = {
      tenant_id: session.user.tenant_id,
    }

    if (clientId) where.client_id = clientId
    if (channel) where.channel = channel

    const messages = await prisma.communication.findMany({
      where,
      include: {
        client: {
          select: {
            id: true,
            name: true,
            photo: true,
            whatsapp: true,
            email: true,
          },
        },
        sent_by: {
          select: {
            id: true,
            name: true,
            avatar: true,
          },
        },
      },
      orderBy: { created_at: 'desc' },
      take: 100,
    })

    return NextResponse.json(messages)
  } catch (error) {
    console.error('Erro ao buscar mensagens:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar mensagens' },
      { status: 500 }
    )
  }
}

// POST - Enviar mensagem
export async function POST(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const body = await req.json()
    const data = messageSchema.parse(body)

    // Buscar cliente se fornecido
    let client = null
    if (data.client_id) {
      client = await prisma.client.findFirst({
        where: {
          id: data.client_id,
          tenant_id: session.user.tenant_id,
        },
      })

      if (!client) {
        return NextResponse.json(
          { error: 'Cliente não encontrado' },
          { status: 404 }
        )
      }
    }

    // Enviar mensagem de acordo com o canal
    let status = 'sent'
    let metadata: any = {}

    switch (data.channel) {
      case 'whatsapp':
        if (client?.whatsapp) {
          const { whatsappService } = await import('@/lib/services/whatsapp')
          const result = await whatsappService.sendText(
            client.whatsapp,
            data.message
          )
          metadata = { whatsapp_message_id: result.key?.id }
        }
        break

      case 'email':
        if (client?.email) {
          const { emailService } = await import('@/lib/services/email')
          await emailService.sendEmail({
            to: client.email,
            subject: data.subject || 'Mensagem',
            text: data.message,
          })
        }
        break

      case 'interno':
        // Chat interno - apenas salvar no banco
        break

      default:
        return NextResponse.json(
          { error: 'Canal não suportado' },
          { status: 400 }
        )
    }

    // Salvar mensagem no banco
    const message = await prisma.communication.create({
      data: {
        tenant_id: session.user.tenant_id,
        client_id: data.client_id,
        channel: data.channel,
        direction: 'outbound',
        subject: data.subject,
        message: data.message,
        attachments: data.attachments,
        status,
        sent_by_id: session.user.id,
        metadata,
      },
      include: {
        client: {
          select: {
            id: true,
            name: true,
            photo: true,
          },
        },
        sent_by: {
          select: {
            id: true,
            name: true,
            avatar: true,
          },
        },
      },
    })

    // TODO: Emitir evento WebSocket para atualização em tempo real
    // await emitNewMessage(message)

    return NextResponse.json(message, { status: 201 })
  } catch (error) {
    if (error instanceof z.ZodError) {
      return NextResponse.json(
        { error: 'Dados inválidos', details: error.errors },
        { status: 400 }
      )
    }

    console.error('Erro ao enviar mensagem:', error)
    return NextResponse.json(
      { error: 'Erro ao enviar mensagem' },
      { status: 500 }
    )
  }
}

'use client'

import { useState, useEffect, useRef } from 'react'
import { Button } from '@/components/ui/button'
import { Input } from '@/components/ui/input'
import { Avatar, AvatarFallback, AvatarImage } from '@/components/ui/avatar'
import { ScrollArea } from '@/components/ui/scroll-area'
import { Send, Paperclip, Smile } from 'lucide-react'
import { useToast } from '@/components/ui/use-toast'
import { format } from 'date-fns'

interface Message {
  id: string
  message: string
  direction: 'inbound' | 'outbound'
  created_at: string
  sent_by?: {
    name: string
    avatar?: string
  }
}

interface ChatInterfaceProps {
  clientId: string
  channel: string
}

export function ChatInterface({ clientId, channel }: ChatInterfaceProps) {
  const { toast } = useToast()
  const [messages, setMessages] = useState<Message[]>([])
  const [newMessage, setNewMessage] = useState('')
  const [isLoading, setIsLoading] = useState(false)
  const scrollRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    loadMessages()
  }, [clientId, channel])

  useEffect(() => {
    scrollToBottom()
  }, [messages])

  async function loadMessages() {
    try {
      const res = await fetch(
        `/api/communication/messages?client_id=${clientId}&channel=${channel}`
      )
      const data = await res.json()
      setMessages(data)
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao carregar mensagens',
        variant: 'destructive',
      })
    }
  }

  async function handleSendMessage() {
    if (!newMessage.trim()) return

    setIsLoading(true)
    try {
      const res = await fetch('/api/communication/messages', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          client_id: clientId,
          channel,
          message: newMessage,
        }),
      })

      if (!res.ok) throw new Error()

      const message = await res.json()
      setMessages([...messages, message])
      setNewMessage('')
    } catch (error) {
      toast({
        title: 'Erro',
        description: 'Erro ao enviar mensagem',
        variant: 'destructive',
      })
    } finally {
      setIsLoading(false)
    }
  }

  function scrollToBottom() {
    if (scrollRef.current) {
      scrollRef.current.scrollTop = scrollRef.current.scrollHeight
    }
  }

  return (
    <div className="flex flex-col h-full">
      {/* Messages Area */}
      <ScrollArea className="flex-1 p-4" ref={scrollRef}>
        <div className="space-y-4">
          {messages.map((message) => (
            <div
              key={message.id}
              className={`flex ${
                message.direction === 'outbound'
                  ? 'justify-end'
                  : 'justify-start'
              }`}
            >
              <div
                className={`flex gap-2 max-w-[70%] ${
                  message.direction === 'outbound'
                    ? 'flex-row-reverse'
                    : 'flex-row'
                }`}
              >
                {message.direction === 'inbound' && (
                  <Avatar className="h-8 w-8">
                    <AvatarImage src="" />
                    <AvatarFallback>C</AvatarFallback>
                  </Avatar>
                )}

                <div>
                  <div
                    className={`rounded-lg p-3 ${
                      message.direction === 'outbound'
                        ? 'bg-blue-500 text-white'
                        : 'bg-gray-100'
                    }`}
                  >
                    {message.message}
                  </div>
                  <div
                    className={`text-xs text-gray-500 mt-1 ${
                      message.direction === 'outbound'
                        ? 'text-right'
                        : 'text-left'
                    }`}
                  >
                    {format(new Date(message.created_at), 'HH:mm')}
                  </div>
                </div>
              </div>
            </div>
          ))}
        </div>
      </ScrollArea>

      {/* Input Area */}
      <div className="border-t p-4">
        <div className="flex gap-2">
          <Button variant="ghost" size="icon">
            <Paperclip className="h-5 w-5" />
          </Button>
          <Button variant="ghost" size="icon">
            <Smile className="h-5 w-5" />
          </Button>
          <Input
            value={newMessage}
            onChange={(e) => setNewMessage(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && handleSendMessage()}
            placeholder="Digite sua mensagem..."
            disabled={isLoading}
          />
          <Button onClick={handleSendMessage} disabled={isLoading}>
            <Send className="h-5 w-5" />
          </Button>
        </div>
      </div>
    </div>
  )
}

  }: {
    amount: number
    currency?: string
    description: string
    metadata?: Record<string, string>
  }) {

    } catch (error) {

      throw error
    }
  }

    lineItems,
    successUrl,
    cancelUrl,
    customerEmail,
    metadata,
  }: {
    lineItems: Array<{
      price_data: {
        currency: string
        product_data: {
          name: string
          description?: string
        }
        unit_amount: number
      }
      quantity: number
    }>
    successUrl: string
    cancelUrl: string
    customerEmail?: string
    metadata?: Record<string, string>
  }) {
    try {
      const session = await stripe.checkout.sessions.create({
        payment_method_types: ['card'],
        line_items: lineItems,
        mode: 'payment',
        success_url: successUrl,
        cancel_url: cancelUrl,
        customer_email: customerEmail,
        metadata,
      })

      return session
    } catch (error) {
      console.error('Erro ao criar Checkout Session Stripe:', error)
      throw error
    }
  }

  async retrievePaymentIntent(id: string) {
    try {
      const paymentIntent = await stripe.paymentIntents.retrieve(id)
      return paymentIntent
    } catch (error) {
      console.error('Erro ao buscar PaymentIntent Stripe:', error)
      throw error
    }
  }

  async refundPayment(paymentIntentId: string, amount?: number) {
    try {
      const refund = await stripe.refunds.create({
        payment_intent: paymentIntentId,
        amount: amount ? Math.round(amount * 100) : undefined,
      })

      return refund
    } catch (error) {
      console.error('Erro ao estornar pagamento Stripe:', error)
      throw error
    }
  }

  verifyWebhookSignature(payload: string, signature: string): any {
    try {
      const event = stripe.webhooks.constructEvent(
        payload,
        signature,
        process.env.STRIPE_WEBHOOK_SECRET!
      )
      return event
    } catch (error) {
      console.error('Erro ao verificar assinatura webhook Stripe:', error)
      throw error
    }
  }
}

import { NextResponse } from 'next/server'
import { prisma } from '@/lib/prisma'
import { mercadoPagoService } from '@/lib/services/mercadopago'

export async function POST(req: Request) {
  try {
    const body = await req.json()

    // Registrar webhook
    await prisma.webhookLog.create({
      data: {
        source: 'mercadopago',
        event: body.type || 'unknown',
        payload: body,
        processed: false,
      },
    })

    // Processar notificação
    if (body.type === 'payment') {
      const paymentId = body.data.id

      // Buscar detalhes do pagamento
      const payment = await mercadoPagoService.getPayment(paymentId)

      // Buscar transação no banco
      const transaction = await prisma.paymentGatewayTransaction.findUnique({
        where: {
          gateway_gateway_id: {
            gateway: 'mercadopago',
            gateway_id: paymentId,
          },
        },
      })

      if (transaction) {
        // Atualizar status
        await prisma.paymentGatewayTransaction.update({
          where: { id: transaction.id },
          data: {
            status: payment.status,
            metadata: payment,
          },
        })

        // Se aprovado, atualizar reserva
        if (payment.status === 'approved' && transaction.reservation_id) {
          const reservation = await prisma.reservation.findUnique({
            where: { id: transaction.reservation_id },
          })

          if (reservation) {
            const newPaidValue =
              Number(reservation.paid_value) + Number(payment.transaction_amount)
            const newBalance = Number(reservation.total_value) - newPaidValue

            await prisma.reservation.update({
              where: { id: transaction.reservation_id },
              data: {
                paid_value: newPaidValue,
                balance: newBalance,
                status: newBalance <= 0 ? 'confirmada' : 'pendente',
              },
            })

            // Registrar pagamento
            await prisma.payment.create({
              data: {
                reservation_id: transaction.reservation_id,
                amount: payment.transaction_amount,
                payment_method: payment.payment_method_id,
                payment_date: new Date(),
                gateway: 'mercadopago',
                gateway_id: paymentId,
                gateway_status: payment.status,
              },

    console.error('Erro ao processar webhook Mercado Pago:', error)

      { error: 'Erro ao processar webhook' },
      { status: 500 }
    )
  }
}

import { NextResponse } from 'next/server'
import { headers } from 'next/headers'
import { prisma } from '@/lib/prisma'
import { stripeService } from '@/lib/services/stripe'

export async function POST(req: Request) {
  try {
    const body = await req.text()
    const signature = headers().get('stripe-signature')

    if (!signature) {
      return NextResponse.json(
        { error: 'Assinatura não encontrada' },
        { status: 400 }
      )
    }

    // Verificar assinatura
    const event = stripeService.verifyWebhookSignature(body, signature)

    // Registrar webhook
    await prisma.webhookLog.create({
      data: {
        source: 'stripe',
        event: event.type,
        payload: event,
        processed: false,
      },
    })

    // Processar evento
    switch (event.type) {
      case 'payment_intent.succeeded':
        const paymentIntent = event.data.object

        // Buscar transação
        const transaction = await prisma.paymentGatewayTransaction.findUnique({
          where: {
            gateway_gateway_id: {
              gateway: 'stripe',
              gateway_id: paymentIntent.id,
            },
          },
        })

        if (transaction && transaction.reservation_id) {
          const reservation = await prisma.reservation.findUnique({
            where: { id: transaction.reservation_id },
          })

          if (reservation) {
            const amount = paymentIntent.amount / 100 // Converter de centavos

            const newPaidValue = Number(reservation.paid_value) + amount
            const newBalance = Number(reservation.total_value) - newPaidValue

            await prisma.reservation.update({
              where: { id: transaction.reservation_id },
              data: {
                paid_value: newPaidValue,
                balance: newBalance,
                status: newBalance <= 0 ? 'confirmada' : 'pendente',
              },
            })

            // Registrar pagamento
            await prisma.payment.create({
              data: {
                reservation_id: transaction.reservation_id,
                amount,
                payment_method: 'card',
                payment_date: new Date(),
                gateway: 'stripe',

    console.error('Erro ao processar webhook Stripe:', error)
    return NextResponse.json(
      { error: 'Erro ao processar webhook' },
      { status: 500 }
    )

import OpenAI from 'openai'

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
})

export class AIService {
  // Previsão de vendas
  async predictSales(historicalData: any[]) {
    try {
      const prompt = `
        Baseado nos dados históricos de vendas abaixo, faça uma previsão
        para os próximos 30 dias. Considere tendências, sazonalidade e
        padrões identificados.
        
        Dados históricos (últimos 6 meses):
        ${JSON.stringify(historicalData, null, 2)}
        
        Retorne a resposta em JSON com o formato:
        {
          "prediction": número estimado de vendas,
          "confidence": percentual de confiança (0-100),
          "factors": ["fator1", "fator2"],
          "recommendations": ["recomendação1", "recomendação2"]
        }
      `

      const response = await openai.chat.completions.create({
        model: 'gpt-4',
        messages: [{ role: 'user', content: prompt }],
        temperature: 0.7,
      })

      const content = response.choices[0].message.content
      return JSON.parse(content || '{}')
    } catch (error) {
      console.error('Erro na previsão de vendas:', error)
      throw error
    }
  }

  // Detecção de churn

## app/api/ai/predict-sales/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { aiService } from '@/lib/services/ai'
import { subMonths, startOfMonth, endOfMonth } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    // Buscar dados históricos (últimos 6 meses)
    const historicalData = []
    for (let i = 5; i >= 0; i--) {
      const date = subMonths(new Date(), i)
      const start = startOfMonth(date)
      const end = endOfMonth(date)

      const reservations = await prisma.reservation.aggregate({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: {
            gte: start,
            lte: end,
          },
        },
        _count: true,
        _sum: {
          total_value: true,
        },
      })

      historicalData.push({
        month: date.toISOString(),
        sales_count: reservations._count,
        revenue: Number(reservations._sum.total_value || 0),
      })
    }

    // Fazer previsão com IA
    const prediction = await aiService.predictSales(historicalData)

    return NextResponse.json({
      historical: historicalData,
      prediction,
    })
  } catch (error) {
    console.error('Erro ao prever vendas:', error)
    return NextResponse.json(
      { error: 'Erro ao prever vendas' },
      { status: 500 }
    )
  }
}

## app/api/ai/churn-detection/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { aiService } from '@/lib/services/ai'
import { differenceInDays } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    // Buscar clientes com última compra há mais de 90 dias
    const clients = await prisma.client.findMany({
      where: {
        tenant_id: session.user.tenant_id,
        is_active: true,
      },
      include: {
        reservations: {
          orderBy: { created_at: 'desc' },
          take: 1,
        },
      },
    })

    const atRiskClients = []

    for (const client of clients) {
      if (client.reservations.length > 0) {
        const lastPurchase = client.reservations[0].created_at
        const daysSinceLastPurchase = differenceInDays(new Date(), lastPurchase)

        if (daysSinceLastPurchase > 90) {
          // Analisar com IA
          const analysis = await aiService.detectChurn({
            client_id: client.id,
            name: client.name,
            days_since_last_purchase: daysSinceLastPurchase,
            total_spent: Number(client.total_spent),
            purchase_count: client.reservations.length,
          })

          if (analysis.risk_level !== 'baixo') {
            atRiskClients.push({
              client,
              analysis,
            })
          }
        }
      }
    }

    return NextResponse.json(atRiskClients)
  } catch (error) {
    console.error('Erro na detecção de churn:', error)
    return NextResponse.json(
      { error: 'Erro na detecção de churn' },
      { status: 500 }
    )
  }
}

## app/api/analytics/sales/route.ts

import { NextResponse } from 'next/server'
import { getServerSession } from 'next-auth'
import { authOptions } from '@/lib/auth'
import { prisma } from '@/lib/prisma'
import { subDays, startOfDay, endOfDay } from 'date-fns'

export async function GET(req: Request) {
  try {
    const session = await getServerSession(authOptions)
    if (!session?.user) {
      return NextResponse.json({ error: 'Não autenticado' }, { status: 401 })
    }

    const { searchParams } = new URL(req.url)
    const days = parseInt(searchParams.get('days') || '30')

    const startDate = startOfDay(subDays(new Date(), days))
    const endDate = endOfDay(new Date())

    // KPIs
    const [
      totalRevenue,
      totalSales,
      newClients,
      avgTicket,
    ] = await Promise.all([
      prisma.reservation.aggregate({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: { gte: startDate, lte: endDate },
        },
        _sum: { total_value: true },
      }),
      prisma.reservation.count({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: { gte: startDate, lte: endDate },
        },
      }),
      prisma.client.count({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: { gte: startDate, lte: endDate },
        },
      }),
      prisma.reservation.aggregate({
        where: {
          tenant_id: session.user.tenant_id,
          created_at: { gte: startDate, lte: endDate },
        },
        _avg: { total_value: true },
      }),
    ])

    // Vendas por origem
    const salesByOrigin = await prisma.client.groupBy({
      by: ['origin'],
      where: {
        tenant_id: session.user.tenant_id,
        created_at: { gte: startDate, lte: endDate },
      },
      _count: true,
    })

    // Top produtos
    const topTrips = await prisma.reservation.groupBy({
      by: ['trip_id'],
      where: {
        tenant_id: session.user.tenant_id,
        created_at: { gte: startDate, lte: endDate },
      },
      _count: true,
      _sum: { total_value: true },
      orderBy: { _count: { trip_id: 'desc' } },
      take: 10,
    })

    // Buscar nomes das viagens
    const tripIds = topTrips.map((t) => t.trip_id)
    const trips = await prisma.trip.findMany({
      where: { id: { in: tripIds } },
      select: { id: true, title: true },
    })

    const topTripsWithNames = topTrips.map((t) => ({
      ...t,
      trip_name: trips.find((trip) => trip.id === t.trip_id)?.title,
    }))

    return NextResponse.json({
      kpis: {
        total_revenue: Number(totalRevenue._sum.total_value || 0),
        total_sales: totalSales,
        new_clients: newClients,
        avg_ticket: Number(avgTicket._avg.total_value || 0),
      },
      sales_by_origin: salesByOrigin,
      top_trips: topTripsWithNames,
    })
  } catch (error) {
    console.error('Erro ao buscar analytics:', error)
    return NextResponse.json(
      { error: 'Erro ao buscar analytics' },
      { status: 500 }
    )
  }
}

## public/sw.js (Service Worker)

const CACHE_NAME = 'visitecrm-v1'
const urlsToCache = [
  '/',
  '/dashboard',
  '/clients',
  '/trips',
  '/offline',
]

// Instalação
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => cache.addAll(urlsToCache))
  )
  self.skipWaiting()
})

// Ativação
self.addEventListener('activate', (event) => {
  event.waitUntil(
    caches.keys().then((cacheNames) => {
      return Promise.all(
        cacheNames.map((cacheName) => {
          if (cacheName !== CACHE_NAME) {
            return caches.delete(cacheName)
          }
        })
      )
    })
  )
  self.clients.claim()
})

// Fetch (Network First, fallback para Cache)
self.addEventListener('fetch', (event) => {
  event.respondWith(
    fetch(event.request)
      .then((response) => {
        const responseClone = response.clone()
        caches.open(CACHE_NAME).then((cache) => {
          cache.put(event.request, responseClone)
        })
        return response
      })
      .catch(() => caches.match(event.request))
  )
})

// Push Notifications
self.addEventListener('push', (event) => {
  const data = event.data.json()

  const options = {
    body: data.body,
    icon: '/icon-192.png',
    badge: '/badge-72.png',
    data: data.url,
  }

  event.waitUntil(
    self.registration.showNotification(data.title, options)
  )
})

// Notification Click
self.addEventListener('notificationclick', (event) => {
  event.notification.close()
  event.waitUntil(clients.openWindow(event.notification.data))
})

## components/pwa/install-prompt.tsx

'use client'

import { useState, useEffect } from 'react'
import { Button } from '@/components/ui/button'
import { X } from 'lucide-react'

export function InstallPrompt() {
  const [deferredPrompt, setDeferredPrompt] = useState<any>(null)
  const [showPrompt, setShowPrompt] = useState(false)

  useEffect(() => {
    const handler = (e: any) => {
      e.preventDefault()
      setDeferredPrompt(e)
      setShowPrompt(true)
    }

    window.addEventListener('beforeinstallprompt', handler)

    return () => {
      window.removeEventListener('beforeinstallprompt', handler)
    }
  }, [])

  const handleInstall = async () => {
    if (!deferredPrompt) return

    deferredPrompt.prompt()
    const { outcome } = await deferredPrompt.userChoice

    if (outcome === 'accepted') {
      console.log('PWA instalado')
    }

    setDeferredPrompt(null)
    setShowPrompt(false)
  }

  if (!showPrompt) return null

  return (
    <div className="fixed bottom-4 left-4 right-4 bg-white p-4 rounded-lg shadow-lg z-50 md:left-auto md:right-4 md:max-w-md">
      <button
        onClick={() => setShowPrompt(false)}
        className="absolute top-2 right-2 text-gray-400 hover:text-gray-600"
      >
        <X className="h-5 w-5" />
      </button>

      <div className="flex items-start gap-4">
        <img src="/icon-192.png" alt="VisiteCRM" className="h-16 w-16 rounded-lg" />
        <div className="flex-1">
          <h3 className="font-bold text-lg">Instalar VisiteCRM</h3>
          <p className="text-sm text-gray-600 mt-1">
            Adicione à tela inicial para acesso rápido e offline
          </p>
          <div className="flex gap-2 mt-3">
            <Button variant="outline" size="sm" onClick={() => setShowPrompt(false)}>
              Depois
            </Button>
            <Button size="sm" onClick={handleInstall}>
              Instalar
            </Button>
          </div>
        </div>
      </div>
    </div>
  )
}
