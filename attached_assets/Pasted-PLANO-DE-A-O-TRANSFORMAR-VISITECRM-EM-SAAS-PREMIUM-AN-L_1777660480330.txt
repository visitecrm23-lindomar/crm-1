import { prisma } from '@visitecrm/database'
import { EventEmitter } from 'events'

export const syncEmitter = new EventEmitter()

// ═══════════════════════════════════════════════════════════
// TIPOS
// ═══════════════════════════════════════════════════════════

type EntityType = 'client' | 'reservation' | 'trip' | 'payment'
type Action = 'create' | 'update' | 'delete'

interface SyncEventData {
  tenantId: string
  entityType: EntityType
  entityId: string
  action: Action
  data: any
}

// ═══════════════════════════════════════════════════════════
// CRIAR EVENTO DE SINCRONIZAÇÃO
// ═══════════════════════════════════════════════════════════

export async function createSyncEvent(event: SyncEventData) {
  // Salvar no banco
  const syncEvent = await prisma.syncEvent.create({
    data: event,
  })

  // Emitir evento em tempo real
  syncEmitter.emit('sync', syncEvent)

  // Processar sincronização
  await processSyncEvent(syncEvent)

  return syncEvent
}

// ═══════════════════════════════════════════════════════════
// PROCESSAR EVENTO DE SINCRONIZAÇÃO
// ═══════════════════════════════════════════════════════════

async function processSyncEvent(event: any) {
  try {
    switch (event.entityType) {
      case 'client':
        await syncClient(event)
        break
      case 'reservation':
        await syncReservation(event)
        break
      case 'trip':
        await syncTrip(event)
        break
      case 'payment':
        await syncPayment(event)
        break
    }

    // Marcar como processado
    await prisma.syncEvent.update({
      where: { id: event.id },
      data: { processed: true },
    })
  } catch (error) {
    console.error('Erro ao processar sync event:', error)
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR CLIENTE
// ═══════════════════════════════════════════════════════════

async function syncClient(event: any) {
  const { entityId, action, data } = event

  if (action === 'update') {
    // Atualizar pipeline se necessário
    if (data.pipelineStage) {
      // Emitir evento para atualizar UI
      syncEmitter.emit('client:pipeline:updated', {
        clientId: entityId,
        stage: data.pipelineStage,
      })
    }

    // Atualizar total gasto se houver nova reserva
    if (data.newReservation) {
      const client = await prisma.client.findUnique({
        where: { id: entityId },
        include: {
          reservations: {
            where: { status: 'PAID' },
          },
        },
      })

      const totalSpent = client?.reservations.reduce(
        (sum, r) => sum + Number(r.paidValue),
        0
      )

      await prisma.client.update({
        where: { id: entityId },
        data: { totalSpent },
      })
    }
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR RESERVA
// ═══════════════════════════════════════════════════════════

async function syncReservation(event: any) {
  const { entityId, action, data } = event

  if (action === 'create' || action === 'update') {
    const reservation = await prisma.reservation.findUnique({
      where: { id: entityId },
      include: {
        client: true,
        trip: true,
      },
    })

    if (!reservation) return

    // Atualizar mapa de assentos da viagem
    await updateTripSeatMap(reservation.tripId, reservation.seats)

    // Atualizar pipeline do cliente
    await updateClientPipeline(reservation.clientId, reservation.status)

    // Atualizar total gasto do cliente
    if (reservation.status === 'PAID') {
      await updateClientTotalSpent(reservation.clientId)
    }

    // Emitir eventos para UI
    syncEmitter.emit('reservation:updated', {
      reservationId: entityId,
      clientId: reservation.clientId,
      tripId: reservation.tripId,
      status: reservation.status,
    })
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR VIAGEM
// ═══════════════════════════════════════════════════════════

async function syncTrip(event: any) {
  const { entityId, action, data } = event

  if (action === 'update') {
    // Recalcular assentos disponíveis
    const trip = await prisma.trip.findUnique({
      where: { id: entityId },
      include: {
        reservations: {
          where: {
            status: { in: ['CONFIRMED', 'PAID', 'IN_PROGRESS'] },
          },
        },
      },
    })

    if (trip) {
      const occupiedSeats = trip.reservations.flatMap((r) => r.seats)
      const availableSeats = trip.totalSeats - occupiedSeats.length

      // Emitir evento
      syncEmitter.emit('trip:seats:updated', {
        tripId: entityId,
        occupiedSeats,
        availableSeats,
      })
    }
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR PAGAMENTO
// ═══════════════════════════════════════════════════════════

async function syncPayment(event: any) {
  const { entityId, action, data } = event

  if (action === 'create' || action === 'update') {
    const payment = await prisma.payment.findUnique({
      where: { id: entityId },
      include: {
        reservation: true,
      },
    })

    if (!payment) return

    // Recalcular saldo da reserva
    const reservation = await prisma.reservation.findUnique({
      where: { id: payment.reservationId },
      include: { payments: true },
    })

    if (reservation) {
      const paidValue = reservation.payments
        .filter((p) => p.status === 'PAID')
        .reduce((sum, p) => sum + Number(p.amount), 0)

      const balance = Number(reservation.totalValue) - paidValue

      await prisma.reservation.update({
        where: { id: payment.reservationId },
        data: {
          paidValue,
          balance,
          status: balance <= 0 ? 'PAID' : 'PENDING',
        },
      })

      // Emitir evento
      syncEmitter.emit('payment:updated', {
        paymentId: entityId,
        reservationId: payment.reservationId,
        paidValue,
        balance,
      })
    }
  }
}

// ═══════════════════════════════════════════════════════════
// HELPERS
// ═══════════════════════════════════════════════════════════

async function updateTripSeatMap(tripId: string, seats: number[]) {
  const trip = await prisma.trip.findUnique({ where: { id: tripId } })
  if (!trip) return

  const seatMap = trip.seatMap as any
  seats.forEach((seat) => {
    seatMap[seat] = { occupied: true }
  })

  await prisma.trip.update({
    where: { id: tripId },
    data: { seatMap },
  })
}

async function updateClientPipeline(
  clientId: string,
  reservationStatus: string
) {
  const stageMap: Record<string, string> = {
    DRAFT: 'NOVO',
    PENDING: 'RESERVADO',
    CONFIRMED: 'RESERVADO',
    PAID: 'RESERVADO',
    IN_PROGRESS: 'EM_VIAGEM',
    COMPLETED: 'POS_VENDA',
    CANCELLED: 'PERDIDO',
  }

  const pipelineStage = stageMap[reservationStatus] || 'NOVO'

  await prisma.client.update({
    where: { id: clientId },
    data: { pipelineStage },
  })
}

async function updateClientTotalSpent(clientId: string) {
  const client = await prisma.client.findUnique({
    where: { id: clientId },
    include: {
      reservations: {
        where: { status: 'PAID' },
      },
    },
  })

  if (!client) return

  const totalSpent = client.reservations.reduce(
    (sum, r) => sum + Number(r.paidValue),
    0
  )

  await prisma.client.update({
    where: { id: clientId },
    data: { totalSpent },
  })
}

import { prisma } from '@visitecrm/database'
import { createSyncEvent } from '@visitecrm/sync'

interface CreateReservationInput {
  tenantId: string
  clientId: string
  tripId: string
  passengers: Array<{
    name: string
    cpf?: string
    birthDate?: string
    ageCategory: string
  }>
  seats: number[]
  totalValue: number
  source: 'dashboard' | 'loja' | 'api'
  createdById?: string
}

export async function createReservation(input: CreateReservationInput) {
  // 1. Validar disponibilidade de assentos
  const trip = await prisma.trip.findUnique({
    where: { id: input.tripId },
    include: {
      reservations: {
        where: {
          status: { in: ['CONFIRMED', 'PAID', 'IN_PROGRESS'] },
        },
      },
    },
  })

  if (!trip) throw new Error('Viagem não encontrada')

  const occupiedSeats = trip.reservations.flatMap((r) => r.seats)
  const hasConflict = input.seats.some((seat) => occupiedSeats.includes(seat))

  if (hasConflict) {
    throw new Error('Alguns assentos já estão ocupados')
  }

  // 2. Criar reserva
  const reservation = await prisma.reservation.create({
    data: {
      tenantId: input.tenantId,
      clientId: input.clientId,
      tripId: input.tripId,
      passengers: input.passengers,
      seats: input.seats,
      totalValue: input.totalValue,
      paidValue: 0,
      balance: input.totalValue,
      status: 'PENDING',
      source: input.source,
      createdById: input.createdById,
      voucherCode: generateVoucherCode(),
    },
  })

  // 3. Criar evento de sincronização
  await createSyncEvent({
    tenantId: input.tenantId,
    entityType: 'reservation',
    entityId: reservation.id,
    action: 'create',
    data: reservation,
  })

  // 4. Atualizar cliente
  await prisma.client.update({
    where: { id: input.clientId },
    data: {
      pipelineStage: 'RESERVADO',
    },
  })

  // 5. Criar evento de sincronização do cliente
  await createSyncEvent({
    tenantId: input.tenantId,
    entityType: 'client',
    entityId: input.clientId,
    action: 'update',
    data: { pipelineStage: 'RESERVADO', newReservation: true },
  })

  return reservation
}

function generateVoucherCode(): string {
  return `VTC-${Date.now()}-${Math.random().toString(36).substr(2, 9).toUpperCase()}`
}

    }

    eventSource.onerror = () => {
      setIsConnected(false)
    }

    eventSource.addEventListener('sync', (event) => {
      const data = JSON.parse(event.data)
      handleSyncEvent(data)
    })

    return () => {
      eventSource.close()
    }
  }, [])

  function handleSyncEvent(event: any) {
    const { entityType, entityId, action } = event

    // Invalidar queries relacionadas
    switch (entityType) {
      case 'client':
        queryClient.invalidateQueries({ queryKey: ['clients'] })
        queryClient.invalidateQueries({ queryKey: ['client', entityId] })
        queryClient.invalidateQueries({ queryKey: ['pipeline'] })
        break

      case 'reservation':
        queryClient.invalidateQueries({ queryKey: ['reservations'] })
        queryClient.invalidateQueries({ queryKey: ['reservation', entityId] })
        queryClient.invalidateQueries({ queryKey: ['clients'] })
        queryClient.invalidateQueries({ queryKey: ['trips'] })
        break

      case 'trip':
        queryClient.invalidateQueries({ queryKey: ['trips'] })
        queryClient.invalidateQueries({ queryKey: ['trip', entityId] })
        break

      case 'payment':
        queryClient.invalidateQueries({ queryKey: ['payments'] })
        queryClient.invalidateQueries({ queryKey: ['reservations'] })
        break
    }
  }

  return { isConnected }
}

import { prisma } from '@visitecrm/database'
import { SyncEventData, EntityType } from '@visitecrm/types'
import { EventEmitter } from 'events'

export const syncEmitter = new EventEmitter()

// ═══════════════════════════════════════════════════════════
// CRIAR EVENTO DE SINCRONIZAÇÃO
// ═══════════════════════════════════════════════════════════

export async function createSyncEvent(event: SyncEventData) {
  const syncEvent = await prisma.syncEvent.create({
    data: event,
  })

  syncEmitter.emit('sync', syncEvent)

  await processSyncEvent(syncEvent)

  return syncEvent
}

// ═══════════════════════════════════════════════════════════
// PROCESSAR EVENTO
// ═══════════════════════════════════════════════════════════

async function processSyncEvent(event: any) {
  try {
    switch (event.entityType as EntityType) {
      case 'client':
        await syncClient(event)
        break
      case 'reservation':
        await syncReservation(event)
        break
      case 'trip':
        await syncTrip(event)
        break
      case 'payment':
        await syncPayment(event)
        break
    }

    await prisma.syncEvent.update({
      where: { id: event.id },
      data: { processed: true },
    })
  } catch (error) {
    console.error('Erro ao processar sync event:', error)
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR CLIENTE
// ═══════════════════════════════════════════════════════════

async function syncClient(event: any) {
  const { entityId, action, data } = event

  if (action === 'update') {
    if (data.pipelineStage) {
      syncEmitter.emit('client:pipeline:updated', {
        clientId: entityId,
        stage: data.pipelineStage,
      })
    }

    if (data.newReservation) {
      const client = await prisma.client.findUnique({
        where: { id: entityId },
        include: {
          reservations: {
            where: { status: 'PAID' },
          },
        },
      })

      if (client) {
        const totalSpent = client.reservations.reduce(
          (sum, r) => sum + Number(r.paidValue),
          0
        )

        await prisma.client.update({
          where: { id: entityId },
          data: { totalSpent },
        })
      }
    }
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR RESERVA
// ═══════════════════════════════════════════════════════════

async function syncReservation(event: any) {
  const { entityId, action } = event

  if (action === 'create' || action === 'update') {
    const reservation = await prisma.reservation.findUnique({
      where: { id: entityId },
      include: {
        client: true,
        trip: true,
      },
    })

    if (!reservation) return

    await updateTripSeatMap(reservation.tripId, reservation.seats)
    await updateClientPipeline(reservation.clientId, reservation.status)

    if (reservation.status === 'PAID') {
      await updateClientTotalSpent(reservation.clientId)
    }

    syncEmitter.emit('reservation:updated', {
      reservationId: entityId,
      clientId: reservation.clientId,
      tripId: reservation.tripId,
      status: reservation.status,
    })
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR VIAGEM
// ═══════════════════════════════════════════════════════════

async function syncTrip(event: any) {
  const { entityId } = event

  const trip = await prisma.trip.findUnique({
    where: { id: entityId },
    include: {
      reservations: {
        where: {
          status: { in: ['CONFIRMED', 'PAID', 'IN_PROGRESS'] },
        },
      },
    },
  })

  if (trip) {
    const occupiedSeats = trip.reservations.flatMap((r) => r.seats)
    const availableSeats = trip.totalSeats - occupiedSeats.length

    syncEmitter.emit('trip:seats:updated', {
      tripId: entityId,
      occupiedSeats,
      availableSeats,
    })
  }
}

// ═══════════════════════════════════════════════════════════
// SINCRONIZAR PAGAMENTO
// ═══════════════════════════════════════════════════════════

async function syncPayment(event: any) {
  const { entityId } = event

  const payment = await prisma.payment.findUnique({
    where: { id: entityId },
    include: {
      reservation: {
        include: { payments: true },
      },
    },
  })

  if (!payment) return

  const reservation = payment.reservation

  const paidValue = reservation.payments
    .filter((p) => p.status === 'PAID')
    .reduce((sum, p) => sum + Number(p.amount), 0)

  const balance = Number(reservation.totalValue) - paidValue

  await prisma.reservation.update({
    where: { id: payment.reservationId },
    data: {
      paidValue,
      balance,
      status: balance <= 0 ? 'PAID' : 'PENDING',
    },
  })

  syncEmitter.emit('payment:updated', {
    paymentId: entityId,
    reservationId: payment.reservationId,
    paidValue,
    balance,
  })
}

async function updateTripSeatMap(tripId: string, seats: number[]) {
  const trip = await prisma.trip.findUnique({ where: { id: tripId } })
  if (!trip) return

  const seatMap = trip.seatMap as any
  seats.forEach((seat) => {
    seatMap[seat] = { occupied: true }
  })

  await prisma.trip.update({
    where: { id: tripId },
    data: { seatMap },
  })
}

async function updateClientPipeline(clientId: string, reservationStatus: string) {
  const stageMap: Record<string, string> = {
    DRAFT: 'NOVO',
    PENDING: 'RESERVADO',
    CONFIRMED: 'RESERVADO',
    PAID: 'RESERVADO',
    IN_PROGRESS: 'EM_VIAGEM',
    COMPLETED: 'POS_VENDA',
    CANCELLED: 'PERDIDO',
  }

  const pipelineStage = stageMap[reservationStatus] || 'NOVO'

  await prisma.client.update({
    where: { id: clientId },
    data: { pipelineStage: pipelineStage as any },
  })
}

async function updateClientTotalSpent(clientId: string) {
  const client = await prisma.client.findUnique({
    where: { id: clientId },
    include: {
      reservations: {
        where: { status: 'PAID' },
      },
    },
  })

  if (!client) return

  const totalSpent = client.reservations.reduce(
    (sum, r) => sum + Number(r.paidValue),
    0
  )

  await prisma.client.update({
    where: { id: clientId },
    data: { totalSpent },
  })
}

export async function createReservation(input: CreateReservationInput) {
  // 1. Validar disponibilidade de assentos
  const trip = await prisma.trip.findUnique({
    where: { id: input.tripId },
    include: {
      reservations: {
        where: {
          status: { in: ['CONFIRMED', 'PAID', 'IN_PROGRESS'] },
        },
      },
    },
  })

  if (!trip) throw new Error('Viagem não encontrada')

  const occupiedSeats = trip.reservations.flatMap((r) => r.seats)
  const hasConflict = input.seats.some((seat) => occupiedSeats.includes(seat))

  if (hasConflict) {
    throw new Error('Alguns assentos já estão ocupados')
  }

  // 2. Criar reserva
  const reservation = await prisma.reservation.create({
    data: {
      tenantId: input.tenantId,
      clientId: input.clientId,
      tripId: input.tripId,
      passengers: input.passengers,
      seats: input.seats,
      totalValue: input.totalValue,
      paidValue: 0,
      balance: input.totalValue,
      status: 'PENDING',
      source: input.source,
      createdById: input.createdById,
      voucherCode: generateVoucherCode(),
    },
  })

  // 3. Sincronizar reserva
  await createSyncEvent({
    tenantId: input.tenantId,
    entityType: 'reservation',
    entityId: reservation.id,
    action: 'create',
    data: reservation,
  })

  // 4. Atualizar cliente
  await prisma.client.update({
    where: { id: input.clientId },
    data: {
      pipelineStage: 'RESERVADO',
    },
  })

  // 5. Sincronizar cliente
  await createSyncEvent({
    tenantId: input.tenantId,
    entityType: 'client',
    entityId: input.clientId,
    action: 'update',
    data: { pipelineStage: 'RESERVADO', newReservation: true },
  })

  return reservation
}

function generateVoucherCode(): string {
  return `VTC-${Date.now()}-${Math.random().toString(36).substr(2, 9).toUpperCase()}`
}

import { Toaster } from 'sonner'

  const [queryClient] = useState(
    () =>

        defaultOptions: {
          queries: {
            staleTime: 0,
            refetchOnWindowFocus: true,
          },
        },
      })
  )

    <SessionProvider>

        <Toaster position="top-right" richColors />
      </QueryClientProvider>
    </SessionProvider>
